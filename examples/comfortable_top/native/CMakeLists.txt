cmake_minimum_required(VERSION 3.20)
project(ComfortableTopNative LANGUAGES CXX)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
get_filename_component(CTUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# Build the existing SDK from source; no browser/telemetry concepts enter it.
add_subdirectory("${CTUI_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sdk" EXCLUDE_FROM_ALL)
pybind11_add_module(_ctui_top_native provider.cpp)
target_link_libraries(_ctui_top_native PRIVATE comfortable_cpu_frames)
target_compile_features(_ctui_top_native PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(_ctui_top_native PRIVATE comfortable_frames)
endif()
if(MSVC)
    target_compile_options(_ctui_top_native PRIVATE /W4 /utf-8 /EHsc)
endif()
