cmake_minimum_required(VERSION 3.20)
project(ComfortableInprocessExample LANGUAGES CXX)
find_package(ComfortableFrames 1 CONFIG REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)

pybind11_add_module(_ctui_inprocess_example inprocess_provider.cpp)
target_link_libraries(_ctui_inprocess_example PRIVATE Comfortable::comfortable_frames)
if(MSVC)
    target_compile_options(_ctui_inprocess_example PRIVATE /W4 /utf-8 /EHsc)
endif()
