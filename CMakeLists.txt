cmake_minimum_required(VERSION 3.20)
project(ComfortableFrames VERSION 1.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_library(comfortable_cpu_frames STATIC native/canvas/cpu_frame.cpp)
target_include_directories(comfortable_cpu_frames PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/canvas/include> $<INSTALL_INTERFACE:include>)
target_compile_features(comfortable_cpu_frames PUBLIC cxx_std_17)
target_link_libraries(comfortable_cpu_frames PUBLIC Threads::Threads)
set_target_properties(comfortable_cpu_frames PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
    target_compile_definitions(comfortable_cpu_frames PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    target_link_libraries(comfortable_cpu_frames PUBLIC rt)
endif()
option(COMFORTABLE_TUI_BUILD_BROWSER "Build the optional CEF service (set CEF_ROOT)" OFF)
option(COMFORTABLE_TUI_BUILD_LINUX_HOST "Withdrawn Linux display helper (unsupported)" OFF)
if(COMFORTABLE_TUI_BUILD_LINUX_HOST)
    message(FATAL_ERROR "Linux graphical support is suspended: WindowHost is not an external terminal overlay. See docs/linux-support-decision.md")
endif()
if(COMFORTABLE_TUI_BUILD_BROWSER AND NOT WIN32)
    message(FATAL_ERROR "The supported browser/display runtime currently requires Windows. Linux graphical support is suspended; see docs/linux-support-decision.md")
endif()
if(WIN32)
add_library(comfortable_frames STATIC native/canvas/frame_provider.cpp)
target_include_directories(comfortable_frames PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/canvas/include> $<INSTALL_INTERFACE:include>)
target_compile_features(comfortable_frames PUBLIC cxx_std_17)
target_compile_definitions(comfortable_frames PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(comfortable_frames PUBLIC d3d11 dxgi bcrypt)
add_library(comfortable_compositor STATIC native/canvas/compositor.cpp)
target_link_libraries(comfortable_compositor PUBLIC comfortable_frames dcomp d3dcompiler user32)
add_library(comfortable_glass STATIC native/glass/glass.cpp)
target_link_libraries(comfortable_glass PUBLIC comfortable_compositor d2d1 dwrite dwmapi ole32 oleacc oleaut32)
add_executable(ctui_gpu_producer native/canvas/producer.cpp)
target_link_libraries(ctui_gpu_producer PRIVATE comfortable_frames)
add_executable(ctui_glass_signal examples/glass_workspace/provider.cpp)
target_link_libraries(ctui_glass_signal PRIVATE comfortable_frames d3dcompiler)
add_executable(ctui_cpu_producer native/canvas/cpu_producer.cpp)
target_link_libraries(ctui_cpu_producer PRIVATE comfortable_frames)
add_executable(ctui_frames_test native/canvas/tests.cpp)
target_include_directories(ctui_frames_test PRIVATE native/canvas)
target_link_libraries(ctui_frames_test PRIVATE comfortable_frames)
enable_testing()
add_test(NAME native_gpu_protocol COMMAND ctui_frames_test)
if(MSVC)
    foreach(target comfortable_frames comfortable_compositor comfortable_glass ctui_gpu_producer ctui_glass_signal)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
endif()
endif() # WIN32: existing shared GPU SDK stays unchanged.
if(COMFORTABLE_TUI_BUILD_BROWSER)
    add_subdirectory(native/browser)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(native/canvas/ComfortableFramesConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/ComfortableFramesConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ComfortableFrames)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ComfortableFramesConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS comfortable_cpu_frames EXPORT ComfortableFramesTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(WIN32)
    install(TARGETS comfortable_frames comfortable_compositor comfortable_glass ctui_gpu_producer EXPORT ComfortableFramesTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(DIRECTORY native/canvas/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ComfortableFramesTargets NAMESPACE Comfortable:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ComfortableFrames)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ComfortableFramesConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ComfortableFramesConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ComfortableFrames)
