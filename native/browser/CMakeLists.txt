# Optional upper-layer service. The frame transport library does not depend on CEF.
enable_language(C)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "BrowserService currently targets Windows and Linux")
endif()
if(NOT IS_DIRECTORY "${CEF_ROOT}/include")
    message(FATAL_ERROR "Set CEF_ROOT to an official CEF 152 minimal distribution")
endif()
file(TO_CMAKE_PATH "${CEF_ROOT}" CEF_ROOT)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(CEF_RUNTIME_LIBRARY_FLAG "/MD" CACHE STRING "CEF wrapper CRT")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} ${CMAKE_CURRENT_BINARY_DIR}/libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
if(WIN32)
    add_library(ctui_browser SHARED browser_service.cpp)
    set_target_properties(ctui_browser PROPERTIES PREFIX "" OUTPUT_NAME "ctui_browser")
else()
    add_executable(ctui_browser browser_service.cpp)
    set_target_properties(ctui_browser PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
SET_COMMON_TARGET_PROPERTIES(ctui_browser)
if(MSVC)
    target_compile_options(ctui_browser PRIVATE /EHsc)
else()
    target_compile_options(ctui_browser PRIVATE -fexceptions -Wno-error)
endif()
target_compile_features(ctui_browser PRIVATE cxx_std_17)
target_include_directories(ctui_browser PRIVATE "${CEF_ROOT}")
target_link_libraries(ctui_browser PRIVATE libcef_lib libcef_dll_wrapper comfortable_cpu_frames ${CEF_STANDARD_LIBS})
target_compile_definitions(ctui_browser PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(WIN32)
    add_custom_command(TARGET ctui_browser POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/Release/bootstrapc.exe" "$<TARGET_FILE_DIR:ctui_browser>/ctui_browser.exe")
endif()
add_custom_command(TARGET ctui_browser POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release" "$<TARGET_FILE_DIR:ctui_browser>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Resources" "$<TARGET_FILE_DIR:ctui_browser>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/LICENSE.txt" "$<TARGET_FILE_DIR:ctui_browser>/CEF-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/CREDITS.html" "$<TARGET_FILE_DIR:ctui_browser>/CREDITS.html")
